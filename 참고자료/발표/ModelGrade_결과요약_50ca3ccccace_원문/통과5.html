```python
def can_attempt(attempts: int, limit: int) -> bool:
    """응시 횟수가 한도 미만인지 확인하여 가능 여부(True/False)를 반환합니다.

    입력값을 직접 변경하지 않고 비교 결과만 반환합니다.
    """
    return attempts < limit


# --- 사용 예시 ---
exam_limit = 3

# 1. 한도 미만인 경우 (2회 응시 / 한도 3회 -> 허용: True)

result1 = can_attempt(current_attempts, exam_limit)
print(f"응시 횟수: {current_attempts}, 한도: {exam_limit} -> 응시 가능: {result1}")

# 2. 한도에 도달한 경우 (3회 응시 / 한도 3회 -> 불가: False)
current_attempts = 3
result2 = can_attempt(current_attempts, exam_limit)
print(f"응시 횟수: {current_attempts}, 한도: {exam_limit} -> 응시 가능: {result2}")

# 3. 한도를 초과한 경우 (4회 응시 / 한도 3회 -> 불가: False)
current_attempts = 4
result3 = can_attempt(current_attempts, exam_limit)
print(f"응시 횟수: {current_attempts}, 한도: {exam_limit} -> 응시 가능: {result3}")
```

```python
def deduplicate_by_name(items):
    seen_names = set()  # 이미 확인한 이름을 기록해 둘 집합(set)
    result = []  # 첫 번째 항목들만 담을 새로운 리스트

    for item in items:
        name = item["name"]
        # 아직 본 적 없는 이름인 경우에만 결과에 추가
        if name not in seen_names:
            seen_names.add(name)
            result.append(item)

    return result


# --- 사용 예시 ---
original_data = [{"name": "솔", "v": 2}, {"name": "별", "v": 1}, {"name": "솔", "v": 8}]

# 함수 실행 (새로운 리스트 반환)
filtered_data = deduplicate_by_name(original_data)

print("결과:", filtered_data)
print("원본 데이터 유지 확인:", original_data)

```

### 동작 원리 설명

1. **새로운 리스트 생성 (`result = []`)**
기존 목록(`items`)의 요소를 직접 삭제(`remove`, `del` 등)하면 원본 데이터가 훼손되거나 반복문 순회 시 오류가 발생할 수 있습니다. 대신 조건에 맞는 항목만 골라 새 리스트에 담는 방식을 사용하여 원본을 그대로 보존합니다.
2. **순서 보존 (`for` 루프)**
원본 리스트의 앞에서부터 순서대로 하나환씩 꺼내어 검사하므로, 결과 리스트에도 항상 첫 등장 순서가 그대로 유지됩니다.

답변
파이썬의 기본 내장 함수인 `sorted()`를 활용하면 문자열을 숫자로 변환하지 않고 사전식(문자 코드 순서)으로 정렬된 새 리스트를 쉽게 얻을 수 있습니다.

```python
def sort_book_numbers(book_numbers: list[str]) -> list[str]:
    """문자열 도서 번호 목록을 받아 사전식 오름차순으로 정렬된 새 리스트를 반환합니다.

    문자열 상태 그대로 정렬하므로 앞자리의 '0'이 보존됩니다.
    """
    return sorted(book_numbers)


# 사용 예시
if __name__ == "__main__":
    original_books = ["0120", "0045", "0003", "0045A", "0012", "0100"]

    # 함수 호출
    sorted_books = sort_book_numbers(original_books)

    print("정렬 전 원본 목록:", original_books)
    print("정렬 후 새 목록:  ", sorted_books)

    # 출력 결과:
    # 정렬 전 원본 목록: ['0120', '0045', '0003', '0045A', '0012', '0100']
    # 정렬 후 새 목록:   ['0003', '0012', '0045', '0045A', '0100', '0120']

```

### 코드 동작 원리

* **사전식 정렬(Lexicographical Order)**: 파이썬은 문자열을 비교할 때 숫자의 크기(크고 작음)를 보지 않고, 사전에서 단어를 찾듯이 **첫 번째 글자부터 순서대로** 크기를 비교합니다. 따라서 `'0003'`, `'0012'`, `'0045'`와 같이 앞자리부터 차례로 정렬됩니다.
* **앞자리 `0` 보존**: 정수(`int`)로 변환하면 `0045`가 숫자 `45`로 바뀌며 앞의 `0`이 사라집니다. 하지만 위 함수는 문자열(`str`) 상태를 유지한 채 비교하기 때문에 앞자리 `0`이 지워지지 않고 온전히 유지됩니다.
* **새 목록 반환**: `list.sort()` 메서드는 기존 리스트 자체를 변경하지만, 파이썬 내장 함수인 `sorted()`는 원본 리스트는 그대로 두고 **정렬된 완전히 새로운 리스트를 생성하여 반환**합니다.

```python
def is_overlapping(start1: int, end1: int, start2: int, end2: int) -> bool:
    """
    두 예약 구간(분 단위)이 겹치는지 판별합니다.

    """
    return start1 < end2 and start2 < end1
```

### 동작 원리 설명

두 시간대가 겹치지 않는 상황을 먼저 떠올리면 쉽게 이해할 수 있습니다.

1. **첫 번째 예약이 완전히 먼저 끝나는 경우:** 첫 번째 종료 시각이 두 번째 시작 시각보다 작거나 같을 때 (`end1 <= start2`)
2. **두 번째 예약이 완전히 먼저 끝나는 경우:** 두 번째 종료 시각이 첫 번째 시작 시각보다 작거나 같을 때 (`end2 <= start1`)

두 예약이 겹치려면 위 두 상황 중 어느 것에도 해당하지 않아야 합니다. 이를 반대로 뒤집으면 다음 두 조건을 동시에 만족해야 합니다.

* 첫 번째 시작 시각이 두 번째 종료 시각보다 빨라야 합니다 (`start1 < end2`).
* 두 번째 시작 시각이 첫 번째 종료 시각보다 빨라야 합니다 (`start2 < end1`).

### 1. 이야기로 이해하는 물방울의 비밀

1. **따뜻한 방 안 (신나게 뛰어노는 아이들)**
따뜻한 방 안에는 눈에 보이지 않지만 공기 중에 수증기 친구들이 가득해요. 이 친구들은 에너지가 넘쳐서 넓은 운동장을 땀 흘리며 마구 뛰어다니는 개구쟁이 아이들과 같아요. 너무 빠르게 이리저리 날아다녀서 우리 눈에는 보이지 않지요.
2. **차가운 물병의 등장 (운동장에 나타난 차가운 얼음성)**
이때 방 한가운데에 냉장고에서 막 꺼낸 차가운 물병이 쑥 들어왔어요. 마치 햇볕 쨍쨍한 운동장 한가운데에 꽁꽁 언 얼음성이 우뚝 솟아난 것과 같아요.
3. **차가운 벽과의 만남 (으~ 추워!)**
신나게 뛰어놀던 아이들(수증기)이 얼음성(물병) 곁을 쌩 지나가다가 차가운 벽에 톡 부딪혔어요. 순간 너무 추워서 몸이 굳고 에너지가 뚝 떨어지게 됩니다.
4. **겉면에 맺힌 물방울 (서로 꼭 끌어안은 아이들)**
추위에 깜짝 놀란 아이들은 더 이상 혼자 뛰어놀지 못하고, "으, 너무 추워! 서로 붙자!" 하면서 얼음성 벽에 기대어 옹기종기 모여들었어요. 그렇게 흩어져 있던 아이들이 한곳에 꼭 붙어 웅크리다 보니, 마침내 우리 눈에 보이는 또렷한 **물방울**이 되어 물병 겉면에 조롱조롱 맺히게 된 것이랍니다.

```python
def unique_stops(stops: list) -> list:
    seen = set()
    result = []
    for stop in stops:
        if stop not in seen:
            seen.add(stop)
            result.append(stop)
    return result
```

```python
def check_return_status(status):
    """
    반납 상태를 확인하여 결과를 반환합니다.
    - True: "완료"
    - False: "확인 필요"
    - None: "상태 미상"
    - 그 외 입력: ValueError 발생
    """
    if status is True:
        return "완료"
    elif status is False:
        return "확인 필요"
    elif status is None:
        return "상태 미상"
    else:
        raise ValueError(f"유효하지 않은 입력입니다: {status!r} (True, False, None만 허용)")

print(check_return_status(None))   # 출력: 상태 미상

# 다른 입력값 전달 시 ValueError 예외 처리 예시
try:

    print(f"예외 발생: {e}")  # 출력: 예외 발생: 유효하지 않은 입력입니다: '미반납' (True, False, None만 허용)

    check_return_status(0)       # 0은 False와 값이 같지만 identity(is)가 다르므로 거절됨
```

### 관찰 순서에 따른 동작 원리

이 시스템에서 일어난 일은 방문객이 대출 목록을 확인하는 과정과 같습니다.

1. **서버의 값을 수정함 (중앙 장부 갱신)**
* 관리자가 서고로 들어가 중앙 장부의 내용을 최신 상태로 바르게 고쳐 적었습니다. 원본 데이터는 정상적으로 저장되었습니다.


2. **화면에 이전 값이 그대로 남음 (메모지 확인)**
* 방문객이 대출 목록을 요청하자, 사서는 굳이 먼 서고까지 걸어가지 않고 책상 위에 놓여 있던 이전 메모지(캐시)를 그대로 읽어 주었습니다.
* 네트워크 요청을 줄이고 속도를 높이기 위한 캐시 동작 원리 때문에, 중앙 장부가 바뀌었는지 매번 확인하지 않고 가지고 있던 복사본을 보여준 것입니다.


3. **캐시를 비움 (낡은 메모지 폐기)**
* 책상 위의 오래된 메모지를 치우자, 사서는 더 이상 참조할 임시 복사본이 없게 되었습니다.


4. **새 값이 화면에 나타남 (중앙 장부 재확인 및 갱신)**
* 메모지가 없으므로 사서는 직접 서고로 이동해 중앙 장부를 확인했습니다.
* 그곳에 저장되어 있던 최신 내용을 읽어 새로운 메모지를 작성하고, 이를 바탕으로 안내 화면을 갱신했습니다.

### 캐시와 원본 데이터의 핵심 차이

* 원본 데이터(Server/DB)는 모든 데이터의 최종 진실(Single Source of Truth)을 담고 있으며, 영구적이고 정확합니다. 대신 네트워크를 통해 가져오는 데 상대적으로 시간이 걸립니다.
* 캐시(Cache)는 성능 최적화를 위해 원본의 일부를 가까운 곳에 복사해 둔 임시 공간입니다. 속도는 매우 빠르지만, 원본이 바뀌었을 때 자동으로 알려주거나 갱신하는 처리가 따로 되어 있지 않다면 일시적으로 이전 데이터를 가리키는 지연(불일치) 현상이 발생합니다.

개발 과정에서 화면이 갱신되지 않을 때 원본 데이터베이스나 API 저장 결과를 먼저 확인하여 정상 저장을 확인했다면, 그다음 단계로는 브라우저나 네트워크 계층의 캐시가 이전 복사본을 붙잡고 있는지 점검하는 것이 자연스러운 접근입니다.
